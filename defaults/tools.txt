## ffmpeg
ffmpeg -i IN -c:v libx264 -c:a aac OUT.mp4|convert to mp4 (h264+aac)
ffmpeg -i IN -vn -c:a copy OUT.m4a|extract audio, no re-encode
ffmpeg -ss START -to END -i IN -c copy OUT.mp4|trim clip, no re-encode
ffmpeg -i IN -c:v libx264 -crf 28 OUT.mp4|compress video (crf 28)
ffmpeg -i IN -vf fps=10,scale=480:-1 OUT.gif|make gif (10 fps, 480px)
ffmpeg -i IN -vf scale=-2:720 OUT.mp4|resize to 720p
ffprobe -hide_banner IN|show media info
## tar
tar -czf OUT.tar.gz DIR|create .tar.gz
tar -xzf IN.tar.gz|extract .tar.gz
tar -cJf OUT.tar.xz DIR|create .tar.xz
tar --zstd -cf OUT.tar.zst DIR|create .tar.zst
tar -xf IN|extract any archive
tar -tf IN|list archive contents
## zip
zip -r OUT.zip DIR|zip folder recursively
unzip IN.zip -d DIR|unzip into DIR
unzip -l IN.zip|list zip contents
## gzip, xz, zstd
gzip -k IN|gzip, keep original
gzip -d IN.gz|decompress .gz
xz -k IN|xz, keep original
xz -d IN.xz|decompress .xz
zstd -k IN|zstd, keep original
zstd -d IN.zst|decompress .zst
## 7z
7z a OUT.7z DIR|create .7z archive
7z x IN -oDIR|extract into DIR
## bzip2, brotli
bzip2 -k IN|bzip2, keep original
brotli IN|brotli, keeps original
