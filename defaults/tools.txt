## ffmpeg
ffmpeg -i {input} -c:v libx264 -c:a aac {output}.mp4|convert to mp4 (h264+aac)
ffmpeg -i {input} -vn -c:a copy {output}.m4a|extract audio, no re-encode
ffmpeg -ss {start} -to {end} -i {input} -c copy {output}.mp4|trim clip, no re-encode
ffmpeg -i {input} -c:v libx264 -crf 28 {output}.mp4|compress video (crf 28)
ffmpeg -i {input} -vf fps=10,scale=480:-1 {output}.gif|make gif (10 fps, 480px)
ffmpeg -i {input} -vf scale=-2:720 {output}.mp4|resize to 720p
ffprobe -hide_banner {input}|show media info
## tar
tar -czf {output}.tar.gz {dir}|create .tar.gz
tar -xzf {input}.tar.gz|extract .tar.gz
tar -cJf {output}.tar.xz {dir}|create .tar.xz
tar --zstd -cf {output}.tar.zst {dir}|create .tar.zst
tar -xf {input}|extract any archive
tar -tf {input}|list archive contents
## zip
zip -r {output}.zip {dir}|zip folder recursively
unzip {input}.zip -d {dir}|unzip into a folder
unzip -l {input}.zip|list zip contents
## gzip, xz, zstd
gzip -k {input}|gzip, keep original
gzip -d {input}.gz|decompress .gz
xz -k {input}|xz, keep original
xz -d {input}.xz|decompress .xz
zstd -k {input}|zstd, keep original
zstd -d {input}.zst|decompress .zst
## 7z
7z a {output}.7z {dir}|create .7z archive
7z x {input} -o{dir}|extract into a folder
## bzip2, brotli
bzip2 -k {input}|bzip2, keep original
brotli {input}|brotli, keeps original
